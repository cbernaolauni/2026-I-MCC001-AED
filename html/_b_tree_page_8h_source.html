57template <typename Container, typename ObjType>
58Count binary_search(Container &container, Count first, Count last, ObjType &object)
59{
60 if (first >= last)
61 return first;
62 while (first < last)
63 {
64 Count mid = (first + last) / 2;
65 if (object == (ObjType)container[mid])
66 return mid;
67 if (object > (ObjType)container[mid])
68 first = mid + 1;
69 else
70 last = mid;
71 }
72 if (object <= (ObjType)container[first])
73 return first;
74 return last;
75}

509template <typename Traits>
510bt_ErrorCode CBTreePage<Traits>::Insert(const key_type &key, ObjIDType ObjID)
511{
512 Count pos = binary_search(m_Keys, 0, m_KeyCount, key);
513 bt_ErrorCode error = bt_ok;
514
515 if (pos < m_KeyCount && (key_type)m_Keys[pos] == key && m_Unique)
516 return bt_duplicate; // this key is duplicate
517
518 if (!m_SubPages[pos]) // this is a leave
519 {
520 ::insert_at(m_Keys, Node(key, ObjID), pos);
521 NumberOfKeys()++;
522 if (Overflow())
523 return bt_overflow;
524 return bt_ok;
525 }
526 // recursive insertion
527 error = m_SubPages[pos]->Insert(key, ObjID);
528 if (error == bt_overflow)
529 {
530 if (!Redistribute1(pos))
531 SplitChild(pos);
532 if (Overflow()) // Propagate overflow
533 return bt_overflow;
534 return bt_ok;
535 }
536 return bt_ok;
537}

545template <typename Traits>
546bool CBTreePage<Traits>::Redistribute1(Count &pos)
547{
548 if (m_SubPages[pos]->Underflow())
549 {
550 // nkol = Number of keys on left brother, nkor = Number of keys on right brother
551 Count nkol = 0, nkor = 0;
552 // is this the first element or there are more elements on right brother
553 if (pos > 0)
554 nkol = m_SubPages[pos - 1]->NumberOfKeys();
555 if (pos < NumberOfKeys())
556 nkor = m_SubPages[pos + 1]->NumberOfKeys();
557
558 if (nkol > nkor)
559 {
560 if (m_SubPages[pos - 1]->NumberOfKeys() > m_SubPages[pos - 1]->MinNumberOfKeys())
561 RedistributeL2R(pos - 1); // bring elements from left brother
562 else
563 {
564 if (pos == NumberOfKeys())
565 --pos;
566 return false;
567 }
568 }
569 else
570 { // nkol <= nkor
571 if (m_SubPages[pos + 1]->NumberOfKeys() > m_SubPages[pos + 1]->MinNumberOfKeys())
572 RedistributeR2L(pos + 1); // bring elements from right brother
573 else
574 {
575 if (pos == 0)
576 ++pos;
577 return false;
578 }
579 }
580 }
581 else
582 { // it is due to overflow
583 Capacity fcol = GetFreeCellsOnLeft(pos); // Free Cells On Left
584 Capacity fcor = GetFreeCellsOnRight(pos); // Free Cells On Left
585
586 if (!fcol && !fcor && m_SubPages[pos]->IsFull())
587 return false;
588 if (fcol > fcor) // There is more space on left
589 RedistributeR2L(pos);
590 else
591 RedistributeL2R(pos);
592 }
593 return true;
594}

602// it considers two brothers m_SubPages[pos-1] && m_SubPages[pos+1]
603// if it fails the only way is merge !
604template <typename Traits>
605bool CBTreePage<Traits>::Redistribute2(Count pos)
606{
607 assert(pos > 0 && pos < NumberOfKeys());
608 assert(m_SubPages[pos - 1] != nullptr && m_SubPages[pos] != nullptr && m_SubPages[pos + 1] != nullptr);
609 assert(m_SubPages[pos - 1]->Underflow() ||
610 m_SubPages[pos]->Underflow() ||
611 m_SubPages[pos + 1]->Underflow());
612
613 if (m_SubPages[pos - 1]->Underflow())
614 {
615 // Rotate R2L
616 RedistributeR2L(pos + 1);
617 RedistributeR2L(pos);
618 if (m_SubPages[pos - 1]->Underflow())
619 return false;
620 }
621 else if (m_SubPages[pos + 1]->Underflow())
622 {
623 // Rotate L2R
624 RedistributeL2R(pos - 1);
625 RedistributeL2R(pos);
626 if (m_SubPages[pos + 1]->Underflow())
627 return false;
628 }
629 else // The problem is exactly at pos !
630 {
631 // Rotate L2R
632 RedistributeL2R(pos - 1);
633 RedistributeR2L(pos + 1);
634 if (m_SubPages[pos]->Underflow())
635 return false;
636 }
637 return true;
638}

646template <typename Traits>
647void CBTreePage<Traits>::RedistributeR2L(Count pos)
648{
649 BTPage *pSource = m_SubPages[pos];
650 BTPage *pTarget = m_SubPages[pos - 1];
651
652 while (pSource->GetNumberOfKeys() > pSource->MinNumberOfKeys() &&
653 pTarget->GetNumberOfKeys() < pSource->GetNumberOfKeys())
654 {
655 // Move from this page to the down-left page \/
656 ::insert_at(pTarget->m_Keys, m_Keys[pos - 1], pTarget->NumberOfKeys()++);
657 // Move the leftest pointer to the rightest position
658 ::insert_at(pTarget->m_SubPages, pSource->m_SubPages[0], pTarget->NumberOfKeys());
659
660 // Move the leftest element to the root
661 m_Keys[pos - 1] = pSource->m_Keys[0];
662
663 // Remove the leftest element from right page
664 ::remove(pSource->m_Keys, 0);
665 ::remove(pSource->m_SubPages, 0);
666 pSource->NumberOfKeys()--;
667 }

683 pTarget->GetNumberOfKeys() < pSource->GetNumberOfKeys())
684 {
685 // Move from this page to the down-RIGHT page \/

687 // Move the rightest pointer to the leftest position
688 ::insert_at(pTarget->m_SubPages, pSource->m_SubPages[pSource->NumberOfKeys()], 0);

693
694 // Remove the rightest element from left page

706template <typename Traits>
707void CBTreePage<Traits>::SplitChild(Count pos)
708{
709 // FIRST: deciding the second page to split
710 BTPage *pChild1 = nullptr, *pChild2 = nullptr;
711
712 if (pos > 0) // is left page full ?
713 if (m_SubPages[pos - 1]->IsFull())
714 {
715 pChild1 = m_SubPages[pos - 1];
716 pChild2 = m_SubPages[pos--];
717 }
718 if (pos < GetNumberOfKeys()) // is right page full ?
719 if (m_SubPages[pos + 1]->IsFull())
720 {
721 pChild1 = m_SubPages[pos];
722 pChild2 = m_SubPages[pos + 1];
723 }
724
725 // SECOND: copy both pages to a temporal one
726 // Create two tmp vector
727 vector<Node> tmpKeys;
728 vector<BTPage *> tmpSubPages;
729
730 // Prepara el vectpor unificado de las 2 paginas a ser divididas en 3
731 // copy from left child
732 MovePage(pChild1, tmpKeys, tmpSubPages);
733 // copy a key from parent
734 tmpKeys.push_back(m_Keys[pos]);
735
736 // copy from right child
737 MovePage(pChild2, tmpKeys, tmpSubPages);
738
739 BTPage *pChild3 = nullptr;
740 Node oi1, oi2;
741 SplitPageInto3(tmpKeys, tmpSubPages, pChild1, pChild2, pChild3, oi1, oi2);
742
743 // copy the first element to the root
744 m_Keys[pos] = oi1;
745 m_SubPages[pos] = pChild1;
746
747 // copy the second element to the root
748 ::insert_at(m_Keys, oi2, pos + 1);
749 ::insert_at(m_SubPages, pChild2, pos + 1);
750 NumberOfKeys()++;
751
752 m_SubPages[pos + 2] = pChild3;
753}

762void CBTreePage<Traits>::SplitPageInto3(vector<Node> &tmpKeys, vector<BTPage *> &tmpSubPages,
763 BTPage *&pChild1, BTPage *&pChild2, BTPage *&pChild3,
764 Node &oi1, Node &oi2)
765{

767 assert(tmpSubPages.size() >= 9);

769 if (!pChild1)
770 pChild1 = new BTPage(m_MaxKeysForChilds, m_Unique);

772 // Split tmpKeys page into 3 pages
773 // copy 1/3 elements to the first child
774 pChild1->clear();
775 Count nKeys = (tmpKeys.size() - 2) / 3;

779 pChild1->m_Keys[i] = tmpKeys[i];
780 pChild1->m_SubPages[i] = tmpSubPages[i];

787
788 if (!pChild2)
789 pChild2 = new BTPage(m_MaxKeysForChilds, m_Unique);
790 pChild2->clear();
791 // copy 1/3 to the second child
792 nKeys += (tmpKeys.size() - 2) / 3 + 1;
793 Count j = 0;
794 for (; i < nKeys; i++, j++)
795 {
796 pChild2->m_Keys[j] = tmpKeys[i];
797 pChild2->m_SubPages[j] = tmpSubPages[i];
798 pChild2->NumberOfKeys()++;
799 }
800 pChild2->m_SubPages[j] = tmpSubPages[i];
801
802 // copy the second element to the root
803 oi2 = tmpKeys[i++];
804
805 // copy 1/3 to the third child
806 if (!pChild3)
807 pChild3 = new BTPage(m_MaxKeysForChilds, m_Unique);
808 pChild3->clear();
809 nKeys = tmpKeys.size();
810 for (j = 0; i < nKeys; i++, j++)
811 {
812 pChild3->m_Keys[j] = tmpKeys[i];
813 pChild3->m_SubPages[j] = tmpSubPages[i];
814 pChild3->NumberOfKeys()++;

898template <typename Traits>
899bt_ErrorCode CBTreePage<Traits>::Remove(const key_type &key, ObjIDType ObjID)
900{
901 bt_ErrorCode error = bt_ok;
902 Count pos = binary_search(m_Keys, 0, m_KeyCount, key);
903
904 if (pos < NumberOfKeys() && key == m_Keys[pos].key)
905 {
906 // This is a leave: First
907 if (!m_SubPages[pos + 1]) // This is a leave ? FIRST CASE !
908 {
909 ::remove(m_Keys, pos);
910 NumberOfKeys()--;
911 if (Underflow())
912 return bt_underflow;
913 return bt_ok;
914 }
915
916 // We FOUND IT BUT it is NOT a leave ? SECOND CASE !
917 {
918 // Get the first element from right branch
919 Node &rFirstFromRight = m_SubPages[pos + 1]->GetFirstNode();
920 // change with a leave
921 swap(m_Keys[pos], rFirstFromRight);
922 // Remove it from this leave
923
924 // Print(cout);
925 error = m_SubPages[++pos]->Remove(key, ObjID);
926 }
927 }
928 else if (pos == NumberOfKeys()) // it is not here, go by the last branch
929 error = m_SubPages[pos]->Remove(key, ObjID);
930 else if (key <= m_Keys[pos].key)
931 { // = is because identical keys are inserted on left (see Insert)
932 if (m_SubPages[pos])
933 error = m_SubPages[pos]->Remove(key, ObjID);
934 else
935 return bt_nofound;
936 }
937
938 if (error == bt_underflow)
939 {
940 // THIRD CASE: After removing the element we have an underflow
941 // Print(cout);
942 if (TreatUnderflow(pos))
943 return bt_ok;
944 // FOURTH CASE: it was not possible to redistribute -> Merge
945 if (IsRoot() && NumberOfKeys() == 2)
946 return MergeRoot();
947 return Merge(pos);
948 }
949 if (error == bt_nofound)
950 return bt_nofound;
951 return bt_ok;
952}

959template <typename Traits>
960bt_ErrorCode CBTreePage<Traits>::Merge(Count pos)
961{
962 assert(m_SubPages[pos - 1]->NumberOfKeys() +
963 m_SubPages[pos]->NumberOfKeys() +
964 m_SubPages[pos + 1]->NumberOfKeys() ==
965 3 * m_SubPages[pos]->MinNumberOfKeys() - 1);
966
967 // FIRST: Put all the elements into a vector
968 vector<Node> tmpKeys;
969 // tmpKeys.resize(nKeys);
970 vector<BTPage *> tmpSubPages;
971
972 BTPage *pChild1 = m_SubPages[pos - 1];
973 BTPage *pChild2 = m_SubPages[pos];
974 BTPage *pChild3 = m_SubPages[pos + 1];
975
976 MovePage(pChild1, tmpKeys, tmpSubPages);
977 tmpKeys.push_back(m_Keys[pos - 1]);
978 MovePage(pChild2, tmpKeys, tmpSubPages);
979 tmpKeys.push_back(m_Keys[pos]);
980 MovePage(pChild3, tmpKeys, tmpSubPages);
981 pChild3->Destroy();
982
983 // Move 1/2 elements to pChild1
984 Capacity nKeys = pChild1->GetFreeCells();
985 Count i = 0;
986 for (; i < nKeys; i++)
987 {
988 pChild1->m_Keys[i] = tmpKeys[i];
989 pChild1->m_SubPages[i] = tmpSubPages[i];
990 pChild1->NumberOfKeys()++;
991 }
992 pChild1->m_SubPages[i] = tmpSubPages[i];
993
994 m_Keys[pos - 1] = tmpKeys[i];
995 m_SubPages[pos - 1] = pChild1;
996
997 ::remove(m_Keys, pos);
998 ::remove(m_SubPages, pos);
999 NumberOfKeys()--;
1000
1001 nKeys = pChild2->GetFreeCells();
1002 Count j = ++i;
1003 for (i = 0; i < nKeys; i++, j++)
1004 {
1005 pChild2->m_Keys[i] = tmpKeys[j];
1006 pChild2->m_SubPages[i] = tmpSubPages[j];
1007 pChild2->NumberOfKeys()++;
1008 }
1009 pChild2->m_SubPages[i] = tmpSubPages[j];
1010 m_SubPages[pos] = pChild2;
1011
1012 if (Underflow())
1013 return bt_underflow;
1014 return bt_ok;
